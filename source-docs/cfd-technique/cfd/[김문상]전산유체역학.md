PROGRAM CIRCLE
C ----------------------------------------------------------------------------------------------------------------------------------------
C CIRCLE MESH GENERATION
C ----------------------------------------------------------------------------------------------------------------------------------------
C P A R A M E T E R S
C ----------------------------------------------------------------------------------------------------------------------------------------
C X, Y : CARTESIAN COORDINATE
C FNX : THE NUMBER OF GRID ALONG TO X-AXIS IN COMPUTATIONAL DOMAIN
C FNY : THE NUMBER OF GRID ALONG TO Y-AXIS IN COMPUTATIONAL DOMAIN
C RAD1 : THE INNER RADIUS
C RAD2 : THE OUTER RADIUS
C STRCH : THE INCREASING RATE OF MESH
C ----------------------------------------------------------------------------------------------------------------------------------------
C F I L E S
C ----------------------------------------------------------------------------------------------------------------------------------------
C CIRCLE.IN : MESH GEOMETRY INPUT DATA
C CIRCLE.ECH : CONFIRMING THE INPUT DATA
C CIRCLE.PLT : RESULTS FOR TECPLOT ( GRID-OUT )
C ----------------------------------------------------------------------------------------------------------------------------------------
PARAMETER(IDIM=3000, JDIM=3000)
COMMON /MESH/ X(IDIM, JDIM), Y(IDIM, JDIM)
COMMON /GEOMETRY/ FNX, FNY, RAD1, RAD2, STRCH
OPEN(UNIT=1, FILE='CIRCLE.IN')
OPEN(UNIT=2, FILE='CIRCLE.ECH')
OPEN(UNIT=3, FILE='CIRCLE.PLT')
C CALL SUBROUTINES
CALL DATAIN !READ THE INPUT GEOMETRY DATA.
CALL DATAECHO !CONFIRM THE INPUT DATA.
CALL MESHGEN !GENERATE THE MESH GRID.
CALL GRIDOUT !PRINT THE GRID INFORMATION
STOP
END

C ----------------------------------------------------------------------------------------------------------------------------------------
C GENERATE THE MESH GRID.
C ----------------------------------------------------------------------------------------------------------------------------------------
SUBROUTINE MESHGEN
PARAMETER(IDIM=3000, JDIM=3000)
COMMON /MESH/ X(IDIM, JDIM), Y(IDIM, JDIM)
COMMON /GEOMETRY/ FNX, FNY, RAD1, RAD2, STRCH
PAI = 4. * ATAN(1.)
THETA = 2.*PAI/(FNX-1.)
IMAX=FNX
JMAX=FNY
C DISCRETIZE THE INNER & OUTER RADIUS.
DO I=1, IMAX
X(I, 1) = RAD1*COS(-THETA*(I-1.))
Y(I, 1) = RAD1*SIN(-THETA*(I-1.))
X(I, JMAX) = RAD2*COS(-THETA*(I-1.))
Y(I, JMAX) = RAD2*SIN(-THETA*(I-1.))
END DO
C FILL-IN THE INSIDE GRID POINTS.
IF (STRCH.EQ.1.) THEN
C FOR EQUIVALENT DIFFERENCE AMONG THE GRIDS
DO J=2, JMAX-1
DO I=1, IMAX-1
XLENG = X(I,JMAX)-X(I,1)
YLENG = Y(I,JMAX)-Y(I,1)
COEFF1 = (J-1.)/(JMAX-1.)
X(I,J) = X(I,1) + XLENG*COEFF1
Y(I,J) = Y(I,1) + YLENG*COEFF1
END DO
END DO
ELSE
C FOR EQUIVALENT RATIO AMONG THE GRIDS
DO J=2, JMAX-1
DO I=1, IMAX-1
XLENG = X(I,JMAX)-X(I,1)
YLENG = Y(I,JMAX)-Y(I,1)
COEFF2 = 1-STRCH**(J-1.)

## 페이지 32
COEFF3 = 1-STRCH**(JMAX-1.)
X(I,J) = X(I,1) + XLENG*COEFF2/COEFF3
Y(I,J) = Y(I,1) + YLENG*COEFF2/COEFF3
END DO
END DO
END IF
C SETTING THE IMAX BOUNDARY GRID POINTS
DO J=1, JMAX
X(IMAX,J) = X(1,J)
Y(IMAX,J) = Y(1,J)
END DO
RETURN
END
C ----------------------------------------------------------------------------------------------------------------------------------------
C PRINT THE GRID INFORMATION TO 'CIRCLE.PLT' FOR TECPLOT
C ----------------------------------------------------------------------------------------------------------------------------------------
SUBROUTINE GRIDOUT
PARAMETER(IDIM=3000, JDIM=3000)
COMMON /MESH/ X(IDIM, JDIM), Y(IDIM, JDIM)
COMMON /GEOMETRY/ FNX, FNY, RAD1, RAD2, STRCH
IMAX = FNX
JMAX = FNY
WRITE(3,100) IMAX, JMAX
100 FORMAT(4X, 'TITLE = "CIRCLE MESH"'/
1 4X, 'VARIABLES = "X" "Y" '/
2 4X, 'ZONE I=', I3, ',', 'J=', I3, 'F=POINT')
WRITE(3,200) ((X(I,J), Y(I,J), I=1, IMAX), J=1, JMAX)
200 FORMAT(2E13.5)
RETURN
END

subroutine ftcs

common/siz/ imin,imax,iminp,imaxm,nstep,nprint
common/par/ stab,alpha,peak,pi,dx,dt,time
common/sol/ x(25),u(25),ue(25)
dimension uxx(25)
do 12 i=iminp,imaxm
12 uxx(i) = stab*(u(i+1) -2.*u(i) +u(i-1))
do 14 i=iminp,imaxm
14 u(i)= u(i) + uxx(i)
return
end

2) Subroutine for Dufort-Frankel Method
subroutine dufort
c Dufort-Frankel Method
common/siz/ imin,imax,iminp,imaxm,nstep,nprint
common/par/ stab,alpha,peak,pi,dx,dt,time
common/sol/ x(25),u(25),ue(25), uu(25)
common/dfm/ coeff1, coeff2, coeff3
dimension uxx(25), temp(25)
do i=iminp,imaxm
uxx(i) = coeff2*uu(i)+coeff3*(u(i+1)+u(i-1))
end do
do i=imin, imax
uu(i)=u(i)
u(i)= uxx(i) / coeff1
end do
return
end

subroutine laasonen
common/siz/ imin,imax,iminp,imaxm,nstep,nprint
common/par/ stab,alpha,peak,pi,dx,dt,time
common/sol/ x(25),u(25),ue(25), uu(25)
real a(imax,imax), r(imax), xx(imax)
c Initialize the matrix A
do i=1, imax-2
do j=1, imax-2
a(i,j)=0.
end do
end do
c Input the coefficients to the matrix A
do i=1, imax-2
a(i,i)=-(2.*stab+1)
if (i.gt.1) then
a(i,i-1)=stab
end if
if (i.lt.imax-2) then
a(i,i+1)=stab
end if
r(i)=-u(i+1)
end do
c Get the upper triangular form
do k=1,imax-3
do i=k+1, imax-2
coeff4=a(i,k)/a(k,k)
r(i)=r(i)-coeff4*r(k)
do j=k, imax-2
a(i,j)=a(i,j)-coeff4*a(k,j)
end do
end do
end do
c Backward Substitution
do i=1, imax-2
xx(imax-1-i)=r(imax-1-i)
do j=imax-1-i,imax-3
xx(imax-1-i)=xx(imax-1-i)-a(imax-1-i,j+1)*xx(j+1)
end do
xx(imax-1-i)=xx(imax-1-i)/a(imax-1-i,imax-1-i)
end do
c Get the solution
u(1)=0
u(imax)=0
do i=2, imax-1
u(i)=xx(i-1)
end do
return
end
